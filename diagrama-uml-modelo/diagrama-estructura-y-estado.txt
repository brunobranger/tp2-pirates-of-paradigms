class GameState {
    - turnNumber: int
    + switchTurn()
    + isGameOver(): boolean
}

class Board {
    - rows: int
    - cols: int
    + getCellAt(pos: Position): Cell
}

class Cell {
    + isOccupied(): boolean
    + assignPortOwner(player: Player)
}

class Position {
    - row: int
    - col: int
}

enum TerrainType {
    OPEN_SEA
    STORM
    PORT
    FOG
}

class Player {
    - name: String
    - surrendered: boolean
}

' Referencias a elementos modelados en otros diagramas
class Ship

enum Faction {
    CRIMSON_BROTHERHOOD
    ABYSS_CROWN
}

class ResourcePool {
    - gold: int
    - wood: int
}
