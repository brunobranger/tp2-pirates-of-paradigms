@startuml
skinparam ClassBackgroundColor #F9F9F9
skinparam ClassBorderColor #333333

class GameState
class GameController
class Board

class Main {
    + {static} main(args: String[])
}

class GameLoader {
    + loadGame(mapFile: String, fleetFile: String): GameState
    + getAvailableMaps(): List<String>
    + getAvailableFleets(): List<String>
}

class GameConfig {
    + {static} REEF_DAMAGE: int
    + {static} PORT_REPAIR_AMOUNT: int
    + {static} INITIAL_GOLD: int
    + {static} INITIAL_WOOD: int
}

interface GameView {
    + showBoard(state: GameState)
    + showTurnSummary(state: GameState)
    + showAttackResult(result: AttackResult)
    + showMoveResult(ship: Ship, origin: Position, dest: Position)
    + showActionUnavailable(ship: Ship, action: ShipAction, reason: String)
    + showEndTurnFireDamage(ship: Ship, hpLost: int, remaining: int)
    + showEndTurnPortRepair(ship: Ship, hp: int)
    + confirmSurrender(player: Player): boolean
    + askMapFile(): String
    + askDestination(reachable: Set<Position>): Position
    + askTarget(targets: List<Ship>): Ship
}

class ConsoleView {
    - scanner: Scanner
    - boardRenderer: BoardRenderer
}

class BoardRenderer {
    + render(gameState: GameState): String
}

enum ShipAction {
    MOVE
    ATTACK
    REPAIR
    BUY_PORT
    EXTINGUISH_FIRE
    CANCEL
}

GameView <|.. ConsoleView

Main ..> GameLoader : instancia
Main ..> ConsoleView : instancia
Main ..> GameController : ensambla
GameLoader ..> GameState : construye

GameController --> "1" GameView
ConsoleView --> "1" BoardRenderer
ConsoleView ..> ShipAction : retorna
GameView ..> GameState : lee
BoardRenderer ..> Board : lee
@enduml
