@startuml
skinparam ClassBackgroundColor #F9F9F9
skinparam ClassBorderColor #333333

class Player
class Cell
class Board

abstract class Ship {
    - name: String
    - currentHp: int
    - maxHp: int
    + receiveDamage(damage: int): int
    + canMove(): boolean
    + canAttack(): boolean
    + getEffectiveMovement(): int
    + getEffectiveAttack(): int
    + getEffectiveRange(): int
    + isAgile(): boolean
    + isOnFire(): boolean
    + isCarryingTreasure(): boolean
}

class Flagship
class Galleon
class Frigate
class Sloop

interface ShipEffect {
    + tick()
    + isExpired(): boolean
    + modifyMovement(base: int): int
    + modifyAttack(base: int): int
    + preventsAttack(): boolean
    + preventsMovement(): boolean
}

class FireEffect
class ImmobilizedEffect
class ReducedMovementEffect
class ReducedAttackEffect
class LostActionEffect
class FavorableWindEffect

interface Treasure {
    + getType(): TreasureType
    + onPickUp(ship: Ship, player: Player)
}

class DynasticTreasure {
    - isFloating: boolean
    + isCarried(): boolean
    + isCaptured(): boolean
    + dropForSink(position: Position)
    + deliver(portPosition: Position)
}

class FloatingTreasure {
    - turnsUntilDisappear: int
    + tick()
    + isGone(): boolean
}

class CombatSystem {
    + attack(attacker: Ship, defender: Ship): AttackResult
    + isInRange(attacker: Ship, defender: Ship): boolean
}

class AttackResult {
    - damageDealt: int
    - defenderSunk: boolean
}

Ship <|-- Flagship
Ship <|-- Galleon
Ship <|-- Frigate
Ship <|-- Sloop

Treasure <|.. DynasticTreasure
Treasure <|.. FloatingTreasure

ShipEffect <|.. FireEffect
ShipEffect <|.. ImmobilizedEffect
ShipEffect <|.. ReducedMovementEffect
ShipEffect <|.. ReducedAttackEffect
ShipEffect <|.. LostActionEffect
ShipEffect <|.. FavorableWindEffect

Player "1" o--> "*" Ship
Ship "1" o--> "*" ShipEffect
Ship --> "0..1" DynasticTreasure : carries
Cell --> "0..1" Treasure
DynasticTreasure --> "1" Player : owner
CombatSystem --> "1" Board
AttackResult --> "1" Ship : attacker
AttackResult --> "1" Ship : defender
@enduml
