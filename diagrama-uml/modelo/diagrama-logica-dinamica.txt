class GameState
class Player

    + getName(): String
    + apply(gameState: GameState): EventResult
    + getDurationInTurns(): int
}

class Squall

class EventManager {
    - {static} instance: EventManager
    + {static} getInstance(): EventManager
    + checkAndGetEvent(gameState: GameState): GameEvent
    + addEvent(event: GameEvent)
}

    - eventName: String
    - consequences: List<String>
    - affectedEntities: List<String>
}

    + check(state: GameState): Player

}

class SurrenderStrategy

enum VictoryCondition {
    FLAGSHIP_SUNK
    TREASURE_CAPTURED
    SURRENDER
}
